% Job Hunter's CV template, for LuaLaTeX. app/cv/latex.py fills in the
% paper size and the body; everything it puts in the body is escaped text and
% the \cv... commands defined here - nothing a person types runs as LaTeX.
\documentclass[10pt]{article}
\usepackage[@@PAPER@@,top=15mm,bottom=15mm,left=17mm,right=17mm]{geometry}
\usepackage{fontspec}
% Lato for the text; DejaVu Sans for any character Lato doesn't have
% (Greek, Cyrillic and symbol coverage beyond Lato's).
\directlua{luaotfload.add_fallback("cvfallback", {"DejaVuSans.ttf:mode=node;"})}
\setmainfont{Lato-Regular.ttf}[
  BoldFont=Lato-Bold.ttf,
  ItalicFont=Lato-Italic.ttf,
  BoldItalicFont=Lato-BoldItalic.ttf,
  RawFeature={fallback=cvfallback}]
\usepackage{xcolor}
\definecolor{accent}{HTML}{1D3A5C}
\definecolor{muted}{HTML}{5A5F66}
\definecolor{rule}{HTML}{C9D1DB}
\usepackage{enumitem}
\setlist[itemize]{leftmargin=1.05em,itemsep=1.5pt,topsep=2.5pt,parsep=0pt,partopsep=0pt,
  label={\color{muted}\textbullet}}
\usepackage[hidelinks,pdfcreator={Job Hunter},pdfproducer={Job Hunter}]{hyperref}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\pagestyle{empty}
\linespread{1.08}
\raggedright
\frenchspacing

\newcommand{\cvname}[1]{{\fontsize{21}{25}\selectfont\bfseries #1\par}\vspace{2pt}}
\newcommand{\cvheadline}[1]{{\large\color{accent}#1\par}\vspace{3pt}}
\newcommand{\cvcontact}[1]{{\small\color{muted}#1\par}}
\newcommand{\cvsep}{\hspace{0.45em}{\color{rule}|}\hspace{0.45em}}
\newcommand{\cvsection}[1]{%
  \vspace{10pt}{\color{accent}\bfseries\MakeUppercase{#1}}\par
  \vspace{2.5pt}{\color{rule}\hrule height 0.6pt}\vspace{5pt}}
% \cventry{title}{dates}, then \cvwhere{organisation, place} if there is one.
% \rightskip=0pt lets \hfill push the dates right despite \raggedright.
\newcommand{\cventry}[2]{%
  \par\pagebreak[2]\begingroup\rightskip=0pt\relax
  {\bfseries #1}\hfill{\small\color{muted}#2}\par\endgroup}
\newcommand{\cvwhere}[1]{{\color{muted}#1\par}}
\newcommand{\cvgap}{\vspace{7pt}}
